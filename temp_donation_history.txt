export function DonationHistoryContent() {
    const { user } = useAuth();
    const [donations, setDonations] = useState<any[]>([]);
    const [search, setSearch] = useState('');
    const [statusFilter, setStatusFilter] = useState('All');
    const [selectedDonation, setSelectedDonation] = useState<any>(null);
    const [isRefreshing, setIsRefreshing] = useState(false);

    const fetchDonations = async () => {
        if (!user) return;
        setIsRefreshing(true);
        const { data, error } = await supabase
            .from('donations')
            .select('*, deliveries(*, ngos(*))')
            .eq('donor_id', user.uid)
            .order('created_at', { ascending: false });
        
        if (error) {
            console.error("Error fetching donations:", error);
        } else {
            setDonations(data || []);
        }
        setTimeout(() => setIsRefreshing(false), 500);
    };

    useEffect(() => {
        fetchDonations();
        const channel = supabase.channel('history_donations')
            .on('postgres_changes', { event: '*', schema: 'public', table: 'donations', filter: `donor_id=eq.${user.uid}` }, () => fetchDonations())
            .subscribe();
        return () => { supabase.removeChannel(channel); };
    }, [user]);

    const filtered = donations.filter(d => {
        if (statusFilter !== 'All' && d.status !== statusFilter.toLowerCase().replace(' ', '_')) return false;
        if (search) {
            const term = search.toLowerCase();
            return (d.title?.toLowerCase().includes(term) || d.food_category?.toLowerCase().includes(term));
        }
        return true;
    });

    const getStatusConfig = (status: string) => {
        switch(status) {
            case 'available': return { color: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-900/30 dark:border-emerald-800/50', label: 'Available', dot: 'bg-emerald-500' };
            case 'requested': return { color: 'bg-yellow-50 text-yellow-700 border-yellow-200 dark:bg-yellow-900/30 dark:border-yellow-800/50', label: 'Requested', dot: 'bg-yellow-500' };
            case 'assigned': 
            case 'accepted': 
            case 'reserved': return { color: 'bg-blue-50 text-blue-700 border-blue-200 dark:bg-blue-900/30 dark:border-blue-800/50', label: 'Accepted', dot: 'bg-blue-500' };
            case 'volunteer_assigned': return { color: 'bg-indigo-50 text-indigo-700 border-indigo-200 dark:bg-indigo-900/30 dark:border-indigo-800/50', label: 'Volunteer Assigned', dot: 'bg-indigo-500' };
            case 'picked_up': 
            case 'in_transit': return { color: 'bg-orange-50 text-orange-700 border-orange-200 dark:bg-orange-900/30 dark:border-orange-800/50', label: 'Picked Up', dot: 'bg-orange-500' };
            case 'delivered': return { color: 'bg-purple-50 text-purple-700 border-purple-200 dark:bg-purple-900/30 dark:border-purple-800/50', label: 'Delivered', dot: 'bg-purple-500' };
            case 'completed': return { color: 'bg-green-100 text-green-800 border-green-300 dark:bg-green-900/50 dark:border-green-800/50', label: 'Completed', dot: 'bg-green-600' };
            case 'expired': return { color: 'bg-red-50 text-red-700 border-red-200 dark:bg-red-900/30 dark:border-red-800/50', label: 'Expired', dot: 'bg-red-500' };
            case 'cancelled': return { color: 'bg-slate-100 text-slate-700 border-slate-200 dark:bg-slate-800 dark:border-slate-700 dark:text-slate-300', label: 'Cancelled', dot: 'bg-slate-500' };
            default: return { color: 'bg-slate-100 text-slate-700 border-slate-200 dark:bg-slate-800 dark:border-slate-700 dark:text-slate-300', label: (status||'').replace('_', ' '), dot: 'bg-slate-400' };
        }
    };

    const getCategoryIcon = (category: string) => {
        const cat = category?.toLowerCase() || '';
        if (cat.includes('cooked') || cat.includes('prepared')) return <Utensils className="w-6 h-6 text-orange-500" />;
        if (cat.includes('veg') && !cat.includes('non')) return <Leaf className="w-6 h-6 text-emerald-500" />;
        if (cat.includes('non-veg') || cat.includes('meat')) return <Drumstick className="w-6 h-6 text-red-500" />;
        if (cat.includes('bakery') || cat.includes('bread')) return <Croissant className="w-6 h-6 text-amber-500" />;
        if (cat.includes('groceries')) return <ShoppingBag className="w-6 h-6 text-blue-500" />;
        if (cat.includes('dry')) return <Wheat className="w-6 h-6 text-yellow-600" />;
        return <Box className="w-6 h-6 text-slate-500" />;
    };

    return (
        <div className="space-y-6 max-w-[1400px] mx-auto animate-in fade-in duration-500 pb-20">
            {/* HEADER */}
            <div className="flex flex-col lg:flex-row lg:items-end justify-between gap-6 mb-8">
                <div>
                    <h2 className="text-3xl md:text-4xl font-black tracking-tight text-slate-900 dark:text-white">Donations</h2>
                    <p className="text-slate-500 font-medium mt-2 text-sm">Manage food donations, monitor status and respond to NGO requests.</p>
                </div>
                <div className="flex flex-wrap items-center gap-3">
                    <div className="relative">
                        <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
                        <Input 
                            placeholder="Search donations..." 
                            value={search} 
                            onChange={(e) => setSearch(e.target.value)} 
                            className="w-[240px] h-10 pl-9 rounded-xl bg-white dark:bg-[#111113] border-slate-200 dark:border-slate-800 focus-visible:ring-emerald-500 font-medium shadow-sm"
                        />
                    </div>
                    <div className="flex items-center bg-white dark:bg-[#111113] border border-slate-200 dark:border-slate-800 rounded-xl p-1 h-10 shadow-sm">
                        <select 
                            value={statusFilter}
                            onChange={(e) => setStatusFilter(e.target.value)}
                            className="pl-3 pr-6 py-1 text-sm bg-transparent border-none focus:outline-none focus:ring-0 text-slate-700 dark:text-slate-200 font-bold cursor-pointer"
                        >
                            {['All', 'Available', 'Requested', 'Accepted', 'Volunteer Assigned', 'Picked Up', 'Delivered', 'Completed', 'Expired', 'Cancelled'].map(s => (
                                <option key={s} value={s}>{s}</option>
                            ))}
                        </select>
                    </div>
                    <Button variant="outline" className="h-10 px-4 rounded-xl font-bold bg-white dark:bg-[#111113] border-slate-200 dark:border-slate-800 hover:bg-slate-50 dark:hover:bg-slate-900 text-slate-700 dark:text-slate-200 flex items-center gap-2 shadow-sm">
                        <Filter className="w-4 h-4" /> Category
                    </Button>
                    <Button variant="outline" className="h-10 px-4 rounded-xl font-bold bg-white dark:bg-[#111113] border-slate-200 dark:border-slate-800 hover:bg-slate-50 dark:hover:bg-slate-900 text-slate-700 dark:text-slate-200 flex items-center gap-2 shadow-sm">
                        <Calendar className="w-4 h-4" /> Sort
                    </Button>
                    <Button 
                        variant="outline" 
                        size="icon" 
                        onClick={fetchDonations}
                        className={`h-10 w-10 shrink-0 rounded-xl bg-white dark:bg-[#111113] border-slate-200 dark:border-slate-800 hover:bg-slate-50 dark:hover:bg-slate-900 shadow-sm ${isRefreshing ? 'animate-spin' : ''}`}
                    >
                        <RefreshCcw className="w-4 h-4 text-slate-600 dark:text-slate-400" />
                    </Button>
                    <Button className="h-10 px-5 rounded-xl bg-emerald-600 hover:bg-emerald-700 text-white font-bold flex items-center gap-2 shadow-sm">
                        <Plus className="w-4 h-4" /> New Donation
                    </Button>
                </div>
            </div>

            {/* DONATION LIST */}
            <div className="flex flex-col gap-6">
                {filtered.length === 0 ? (
                    <div className="py-32 text-center bg-white dark:bg-[#111113] border border-slate-200 dark:border-slate-800 rounded-[24px] flex flex-col items-center justify-center shadow-sm">
                        <div className="w-20 h-20 mb-6 bg-slate-50 dark:bg-slate-900 rounded-2xl flex items-center justify-center text-slate-400 dark:text-slate-600 border border-slate-100 dark:border-slate-800">
                            <Box className="w-10 h-10" />
                        </div>
                        <h3 className="text-2xl font-black text-slate-900 dark:text-white mb-2">No Donations Yet</h3>
                        <p className="text-slate-500 font-medium mb-8 max-w-md">Create your first food donation to begin helping nearby NGOs.</p>
                        <Button className="h-12 px-8 rounded-xl bg-emerald-600 hover:bg-emerald-700 text-white font-bold shadow-md flex items-center gap-2">
                            <Plus className="w-5 h-5" /> Create Donation
                        </Button>
                    </div>
                ) : filtered.map(d => {
                    const status = getStatusConfig(d.status);
                    return (
                        <motion.div 
                            key={d.id} 
                            whileHover={{ y: -2, scale: 1.01 }}
                            className="group bg-white dark:bg-[#111113] border border-slate-200 dark:border-slate-800 shadow-sm hover:shadow-md transition-all duration-300 rounded-[18px] p-5 flex flex-col md:flex-row md:items-center justify-between gap-6 cursor-pointer"
                            onClick={() => setSelectedDonation(d)}
                        >
                            {/* LEFT */}
                            <div className="flex items-center gap-4 min-w-[280px]">
                                <div className="w-14 h-14 rounded-2xl bg-slate-50 dark:bg-slate-800/50 border border-slate-100 dark:border-slate-800 flex items-center justify-center shrink-0">
                                    {getCategoryIcon(d.food_category)}
                                </div>
                                <div className="flex flex-col">
                                    <h4 className="font-bold text-slate-900 dark:text-white text-base leading-tight group-hover:text-emerald-600 transition-colors line-clamp-1">{d.title || d.food_category || 'Surplus Food'}</h4>
                                    <div className="flex items-center gap-2 text-xs text-slate-500 mt-1.5 font-medium">
                                        <span className="flex items-center gap-1"><Clock className="w-3.5 h-3.5" /> {new Date(d.created_at).toLocaleDateString()}</span>
                                        <div className="w-1 h-1 rounded-full bg-slate-300 dark:bg-slate-700"></div>
                                        <span className="font-mono text-[10px] bg-slate-100 dark:bg-slate-800 px-1.5 py-0.5 rounded text-slate-600 dark:text-slate-400">#{d.id.substring(0,8).toUpperCase()}</span>
                                    </div>
                                </div>
                            </div>

                            {/* CENTER */}
                            <div className="hidden md:flex flex-1 items-center justify-between px-6 border-l border-r border-slate-100 dark:border-slate-800/50">
                                <div className="flex flex-col gap-1.5">
                                    <span className="text-[10px] font-bold text-slate-400 uppercase tracking-wider">Food Category</span>
                                    <span className="text-sm font-bold text-slate-700 dark:text-slate-300 capitalize">{d.food_category?.replace('_', ' ')}</span>
                                </div>
                                <div className="flex flex-col gap-1.5">
                                    <span className="text-[10px] font-bold text-slate-400 uppercase tracking-wider">Weight & Meals</span>
                                    <div className="flex items-center gap-2 text-sm font-bold text-slate-700 dark:text-slate-300">
                                        <span>{d.quantity_kg} {d.unit || 'kg'}</span>
                                        <div className="w-1 h-1 rounded-full bg-slate-300 dark:bg-slate-700"></div>
                                        <span className="text-emerald-600 dark:text-emerald-500">{d.estimated_meals} meals</span>
                                    </div>
                                </div>
                                <div className="flex flex-col gap-1.5">
                                    <span className="text-[10px] font-bold text-slate-400 uppercase tracking-wider">Expiry / Pickup</span>
                                    <div className="flex flex-col text-xs font-semibold text-slate-600 dark:text-slate-400">
                                        <span>Exp: {d.expiry_time ? new Date(d.expiry_time).toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}) : 'N/A'}</span>
                                        <span className="text-[10px] text-slate-400">
                                            Win: {d.pickup_start_time ? new Date(d.pickup_start_time).toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}) : 'Any'}
                                        </span>
                                    </div>
                                </div>
                            </div>

                            {/* RIGHT */}
                            <div className="flex items-center justify-between md:justify-end gap-5 min-w-[240px]">
                                <Badge variant="outline" className={`${status.color} px-2.5 py-1 font-black rounded-lg uppercase tracking-wider text-[10px] flex items-center gap-1.5 border`}>
                                    <div className={`w-1.5 h-1.5 rounded-full ${status.dot}`}></div>
                                    {status.label}
                                </Badge>
                                <div className="flex items-center gap-1">
                                    <Button variant="ghost" size="sm" className="h-8 px-3 rounded-lg font-bold text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50 dark:hover:bg-emerald-900/20 text-xs">
                                        View Details
                                    </Button>
                                    <Button variant="ghost" size="icon" className="h-8 w-8 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-300" onClick={(e) => { e.stopPropagation(); }}>
                                        <MoreVertical className="w-4 h-4" />
                                    </Button>
                                </div>
                            </div>
                        </motion.div>
                    );
                })}
            </div>

            {/* DETAILS DRAWER */}
            <AnimatePresence>
                {selectedDonation && (
                    <>
                        <motion.div 
                            initial={{ opacity: 0 }} 
                            animate={{ opacity: 1 }} 
                            exit={{ opacity: 0 }} 
                            onClick={() => setSelectedDonation(null)}
                            className="fixed inset-0 bg-slate-900/20 backdrop-blur-sm z-40"
                        />
                        <motion.div 
                            initial={{ x: "100%" }} 
                            animate={{ x: 0 }} 
                            exit={{ x: "100%" }} 
                            transition={{ type: "spring", damping: 25, stiffness: 200 }}
                            className="fixed top-0 right-0 h-full w-full sm:w-[500px] bg-white dark:bg-[#09090b] z-50 shadow-2xl border-l border-slate-200 dark:border-slate-800 overflow-y-auto flex flex-col"
                        >
                            {/* DRAWER HEADER */}
                            <div className="flex items-center justify-between p-6 border-b border-slate-100 dark:border-slate-800/60 sticky top-0 bg-white/90 dark:bg-[#09090b]/90 backdrop-blur-md z-10">
                                <div className="flex flex-col gap-1.5">
                                    <div className="flex items-center gap-3">
                                        <h3 className="text-xl font-black text-slate-900 dark:text-white line-clamp-1">{selectedDonation.title || selectedDonation.food_category?.replace('_', ' ')}</h3>
                                        <Badge variant="outline" className={`${getStatusConfig(selectedDonation.status).color} px-2 py-0.5 font-black rounded-md uppercase tracking-wider text-[9px] border`}>
                                            {getStatusConfig(selectedDonation.status).label}
                                        </Badge>
                                    </div>
                                    <div className="flex items-center gap-2 text-xs text-slate-500 font-medium">
                                        <span className="font-mono bg-slate-100 dark:bg-slate-800 px-1.5 py-0.5 rounded text-slate-600 dark:text-slate-400">#{selectedDonation.id.substring(0,8).toUpperCase()}</span>
                                        <span>•</span>
                                        <span>{new Date(selectedDonation.created_at).toLocaleString()}</span>
                                    </div>
                                </div>
                                <div className="flex items-center gap-2 shrink-0">
                                    <Button variant="outline" size="sm" className="h-8 rounded-lg text-xs font-bold bg-white dark:bg-transparent border-slate-200 dark:border-slate-700 text-slate-700 dark:text-slate-300">Edit</Button>
                                    <Button variant="outline" size="sm" className="h-8 rounded-lg text-xs font-bold bg-white dark:bg-transparent border-slate-200 dark:border-slate-700 text-red-600 dark:text-red-500">Cancel</Button>
                                    <Button variant="ghost" size="icon" className="h-8 w-8 rounded-lg text-slate-500 hover:bg-slate-100 dark:hover:bg-slate-800" onClick={() => setSelectedDonation(null)}>
                                        <X className="w-4 h-4" />
                                    </Button>
                                </div>
                            </div>

                            {/* DRAWER CONTENT */}
                            <div className="p-6 space-y-8 pb-24">
                                
                                {/* SECTION 1: Donation Information */}
                                <div className="space-y-3">
                                    <h4 className="text-xs font-bold text-slate-400 uppercase tracking-widest flex items-center gap-2"><Box className="w-3.5 h-3.5 text-emerald-500" /> Donation Information</h4>
                                    <div className="bg-white dark:bg-[#111113] rounded-2xl border border-slate-200 dark:border-slate-800/60 shadow-sm overflow-hidden">
                                        <div className="grid grid-cols-2 divide-x divide-slate-100 dark:divide-slate-800/60 border-b border-slate-100 dark:border-slate-800/60">
                                            <div className="p-4 flex flex-col gap-1.5 bg-slate-50/50 dark:bg-[#151518]/50">
                                                <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Food Category</span>
                                                <span className="text-sm font-bold text-slate-900 dark:text-white capitalize">{selectedDonation.food_category?.replace('_', ' ')}</span>
                                            </div>
                                            <div className="p-4 flex flex-col gap-1.5 bg-slate-50/50 dark:bg-[#151518]/50">
                                                <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Weight / Meals</span>
                                                <span className="text-sm font-bold text-slate-900 dark:text-white">{selectedDonation.quantity_kg} {selectedDonation.unit || 'kg'} • {selectedDonation.estimated_meals} meals</span>
                                            </div>
                                        </div>
                                        <div className="grid grid-cols-2 divide-x divide-slate-100 dark:divide-slate-800/60 border-b border-slate-100 dark:border-slate-800/60">
                                            <div className="p-4 flex flex-col gap-1.5 bg-slate-50/50 dark:bg-[#151518]/50">
                                                <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Packaging Type</span>
                                                <span className="text-sm font-bold text-slate-900 dark:text-white capitalize">{selectedDonation.packaging_type || 'Standard'}</span>
                                            </div>
                                            <div className="p-4 flex flex-col gap-1.5 bg-slate-50/50 dark:bg-[#151518]/50">
                                                <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Expiry</span>
                                                <span className="text-sm font-bold text-slate-900 dark:text-white">{selectedDonation.expiry_time ? new Date(selectedDonation.expiry_time).toLocaleString() : 'N/A'}</span>
                                            </div>
                                        </div>
                                        <div className="p-4 flex flex-col gap-1.5 border-b border-slate-100 dark:border-slate-800/60 bg-slate-50/50 dark:bg-[#151518]/50">
                                            <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Pickup Window</span>
                                            <span className="text-sm font-bold text-slate-900 dark:text-white">
                                                {selectedDonation.pickup_start_time ? new Date(selectedDonation.pickup_start_time).toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}) : 'Any'} - 
                                                {selectedDonation.pickup_end_time ? new Date(selectedDonation.pickup_end_time).toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'}) : 'Any'}
                                            </span>
                                        </div>
                                        {selectedDonation.special_instructions && (
                                            <div className="p-4 flex flex-col gap-1.5 bg-yellow-50/30 dark:bg-yellow-900/10">
                                                <span className="text-[10px] font-bold text-yellow-600 dark:text-yellow-500 uppercase tracking-wider">Special Instructions</span>
                                                <span className="text-sm font-medium text-yellow-800 dark:text-yellow-200 mt-0.5">{selectedDonation.special_instructions}</span>
                                            </div>
                                        )}
                                    </div>
                                </div>

                                {/* SECTION 2: Pickup Information */}
                                <div className="space-y-3">
                                    <h4 className="text-xs font-bold text-slate-400 uppercase tracking-widest flex items-center gap-2"><MapPin className="w-3.5 h-3.5 text-emerald-500" /> Pickup Information</h4>
                                    <div className="bg-white dark:bg-[#111113] rounded-2xl border border-slate-200 dark:border-slate-800/60 shadow-sm p-5 space-y-4">
                                        <div className="flex flex-col gap-1.5">
                                            <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Restaurant Name</span>
                                            <span className="text-sm font-bold text-slate-900 dark:text-white">{selectedDonation.donor_name || 'Restaurant'}</span>
                                        </div>
                                        <div className="flex flex-col gap-1.5">
                                            <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Pickup Address</span>
                                            <span className="text-sm font-bold text-slate-900 dark:text-white">{selectedDonation.pickup_address}</span>
                                        </div>
                                        <div className="grid grid-cols-2 gap-4">
                                            <div className="flex flex-col gap-1.5">
                                                <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Contact Person</span>
                                                <span className="text-sm font-bold text-slate-900 dark:text-white">{selectedDonation.contact_person || 'Manager'}</span>
                                            </div>
                                            <div className="flex flex-col gap-1.5">
                                                <span className="text-[10px] font-bold text-slate-500 uppercase tracking-wider">Phone</span>
                                                <span className="text-sm font-bold text-slate-900 dark:text-white">{selectedDonation.contact_phone || 'N/A'}</span>
                                            </div>
                                        </div>
                                    </div>
                                </div>

                                {/* SECTION 3: NGO Information */}
                                <div className="space-y-3">
                                    <h4 className="text-xs font-bold text-slate-400 uppercase tracking-widest flex items-center gap-2"><Store className="w-3.5 h-3.5 text-emerald-500" /> NGO Information</h4>
                                    <div className="bg-white dark:bg-[#111113] rounded-2xl border border-slate-200 dark:border-slate-800/60 shadow-sm p-5">
                                        {selectedDonation.deliveries?.[0]?.ngos ? (
                                            <div className="flex items-center gap-4">
                                                <div className="w-12 h-12 rounded-xl bg-slate-100 dark:bg-slate-800 flex items-center justify-center text-slate-500">
                                                    <Store className="w-6 h-6" />
                                                </div>
                                                <div className="flex flex-col">
                                                    <span className="text-sm font-black text-slate-900 dark:text-white">{selectedDonation.deliveries[0].ngos.name}</span>
                                                    <span className="text-xs font-bold text-slate-500 mt-0.5">Assigned NGO</span>
                                                </div>
                                            </div>
                                        ) : (
                                            <span className="text-sm font-medium text-slate-500">No NGO assigned yet.</span>
                                        )}
                                    </div>
                                </div>

                                {/* SECTION 4: Volunteer */}
                                <div className="space-y-3">
                                    <h4 className="text-xs font-bold text-slate-400 uppercase tracking-widest flex items-center gap-2"><HeartHandshake className="w-3.5 h-3.5 text-emerald-500" /> Volunteer</h4>
                                    <div className="bg-white dark:bg-[#111113] rounded-2xl border border-slate-200 dark:border-slate-800/60 shadow-sm p-5">
                                        {selectedDonation.deliveries?.[0]?.volunteer_id ? (
                                            <div className="flex items-center gap-4">
                                                <div className="w-12 h-12 rounded-xl bg-indigo-50 dark:bg-indigo-900/30 flex items-center justify-center text-indigo-600 dark:text-indigo-400 border border-indigo-100 dark:border-indigo-800/50">
                                                    <HeartHandshake className="w-6 h-6" />
                                                </div>
                                                <div className="flex flex-col">
                                                    <span className="text-sm font-black text-slate-900 dark:text-white">Volunteer Assigned</span>
                                                    <span className="text-xs font-bold text-slate-500 mt-0.5">Ready for pickup</span>
                                                </div>
                                            </div>
                                        ) : (
                                            <span className="text-sm font-medium text-slate-500">Volunteer not assigned.</span>
                                        )}
                                    </div>
                                </div>

                                {/* SECTION 5: Lifecycle Timeline */}
                                <div className="space-y-3">
                                    <h4 className="text-xs font-bold text-slate-400 uppercase tracking-widest flex items-center gap-2"><History className="w-3.5 h-3.5 text-emerald-500" /> Lifecycle Timeline</h4>
